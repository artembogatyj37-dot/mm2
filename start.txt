local texts, cache, loadedPages = {}, {}, 0

local function normalize(s)
    s = s:gsub("’", "'")
    return (s:lower():gsub("^%s+", ""):gsub("%s+$", ""))
end

local function clean(html)
    html = html:gsub("<[^>]+>", " ")
    html = html:gsub("&amp;", "&"):gsub("&#39;", "'"):gsub("’", "'")
    html = html:gsub("%s+", " ")
    return " " .. html:lower()
end

local function refreshValues()
    local new = {}
    for _, url in ipairs(PAGES) do
        local ok, res = pcall(function() return game:HttpGet(url) end)
        if ok and type(res) == "string" then
            new[#new + 1] = {src = url:match("/([^/]+)$") or "?", text = clean(res)}
        end
        task.wait(1)
    end
    if #new > 0 then
        texts = new
        cache = {}
        loadedPages = #new
    end
end

local function getValue(name)
    local key = normalize(name)
    if cache[key] ~= nil then
        if cache[key] == false then return nil end
        return cache[key][1], cache[key][2]
    end
    local esc = (key:gsub("%p", "%%%0"))
    local pat = " " .. esc .. " value %- ([%d,%.]+)"
    for _, p in ipairs(texts) do
        local m = p.text:match(pat)
        if m then
            local v = tonumber((m:gsub(",", "")))
            if v then
                cache[key] = {v, p.src}
                return v, p.src
            end
        end
    end
    cache[key] = false
    return nil
end

local function isShown(obj, root)
    local o = obj
    while o and o ~= root do
        if o:IsA("GuiObject") and not o.Visible then return false end
        o = o.Parent
    end
    return true
end

local function readName(slot)
    local best
    for _, d in ipairs(slot:GetDescendants()) do
        if d:IsA("TextLabel") and (d.Name == "Label" or d.Name == "ColoredName") then
            local t = d.Text
            if t ~= "" and t ~= "Loading" and t ~= "..." then
                best = t
                if d.Name == "ColoredName" then break end
            end
        end
    end
    return best
end

local function fmt(n)
    local s
    if n % 1 == 0 then s = string.format("%d", n) else s = string.format("%.1f", n) end
    local sign, int, rest = s:match("^(-?)(%d+)(.*)$")
    int = int:reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
    return sign .. int .. rest
end

local function setPrice(slot, v)
    pcall(function()
        local lbl = slot:FindFirstChild("MM2Price")
        if not lbl then
            lbl = Instance.new("TextLabel")
            lbl.Name = "MM2Price"
            lbl.AnchorPoint = Vector2.new(0.5, 1)
            lbl.Size = UDim2.new(0.7, 0, 0, 22)
            lbl.BackgroundColor3 = Color3.fromRGB(20, 40, 80)
            lbl.BackgroundTransparency = 0.35
            lbl.Font = Enum.Font.GothamBold
            lbl.TextSize = 16
            lbl.TextColor3 = Color3.new(1, 1, 1)
            lbl.ZIndex = 50
            Instance.new("UICorner", lbl).CornerRadius = UDim.new(0, 8)
            lbl.Parent = slot
        end
        lbl.Position = UDim2.new(0.5, 0, PRICE_Y, 0)
        lbl.Text = v and fmt(v) or "?"
        lbl.Visible = true
    end)
end

local function hidePrice(slot)
    local pl = slot:FindFirstChild("MM2Price")
    if pl then pl.Visible = false end
end

local function sumOffer(offer)
    local total, dbg = 0, {}
    local cont = offer:FindFirstChild("Container")
    if not cont then return 0, dbg end
    for _, slot in ipairs(cont:GetChildren()) do
        if slot.Name:match("^NewItem") then
            if isShown(slot, offer) then
                local name = readName(slot)
                if name then
                    local v, src = getValue(name)
                    if v then total += v end
                    setPrice(slot, v)
                    dbg[#dbg + 1] = name .. "=" .. (v and (v .. "@" .. src) or "?")
                else
                    hidePrice(slot)
                end
            else
                hidePrice(slot)
            end
        end
    end
    return total, dbg
end

local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function render(a, b)
    local diff = b - a
    local total = a + b
    local ratio = (total == 0) and 0.5 or (b / total)
    ratio = math.clamp(ratio, 0.06, 1)

    local color, text
    if diff > 0 then
        color, text = COLOR_WIN, "WIN +" .. fmt(diff)
    elseif diff < 0 then
        color, text = COLOR_LOSE, "LOSE " .. fmt(diff)
    else
        color, text = COLOR_EVEN, "EVEN"
    end

    title.Text = text
    mineLbl.Text = fmt(a)
    theirsLbl.Text = fmt(b)
    TweenService:Create(fill, tweenInfo, {
        Size = UDim2.new(ratio, 0, 1, 0),
        BackgroundColor3 = color,
    }):Play()
end

local function isVisible(o)
    while o do
        if o:IsA("GuiObject") and not o.Visible then return false end
        if o:IsA("ScreenGui") and not o.Enabled then return false end
        o = o.Parent
    end
    return true
end

local function path(root, ...)
    local o = root
    for _, n in ipairs({...}) do
        if not o then return nil end
        o = o:FindFirstChild(n)
    end
    return o
end

local function getVP()
    return path(playerGui, "MainGUI", "Lobby", "ViewProfile")
end

local function fireSig(sig)
    if firesignal then
        pcall(firesignal, sig)
    elseif getconnections then
        pcall(function()
            for _, c in ipairs(getconnections(sig)) do
                pcall(function() c:Fire() end)
            end
        end)
    end
end

local function clickUntil(b, check)
    if not b then return false end
    local seq = {
        function() fireSig(b.MouseButton1Click) end,
        function() fireSig(b.Activated) end,
        function() fireSig(b.MouseButton1Down); fireSig(b.MouseButton1Up) end,
    }
    for _, f in ipairs(seq) do
        pcall(f)
        for _ = 1, 8 do
            task.wait(0.03)
            local ok, r = pcall(check)
            if ok and r then return true end
        end
    end
    local ok, r = pcall(check)
    return ok and r and true or false
end

local function findClickable(label)
    local o = label
    for _ = 1, 5 do
        if not o then return nil end
        if o:IsA("GuiButton") then return o end
        local b = o:FindFirstChildWhichIsA("GuiButton", true)
        if b then return b end
        o = o.Parent
    end
    return nil
end

local function scanLabels(names)
    local found = {}
    local vp = getVP()
    for _, d in ipairs(playerGui:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local k = d.Text:lower()
            if names[k] and not found[k] and isVisible(d) and not (vp and d:IsDescendantOf(vp)) then
                found[k] = d
            end
        end
    end
    return found
end

local function findLabel(plr, found)
    return found[plr.Name:lower()] or found[plr.DisplayName:lower()]
end

local function findButtonByText(txt)
    local vp = getVP()
    for _, d in ipairs(playerGui:GetDescendants()) do
        if d:IsA("TextButton") and d.Text == txt and isVisible(d) and not (vp and d:IsDescendantOf(vp)) then
            return d
        end
    end
    for _, d in ipairs(playerGui:GetDescendants()) do
        if d:IsA("TextLabel") and d.Text == txt and isVisible(d) and not (vp and d:IsDescendantOf(vp)) then
            local c = findClickable(d)
            if c then return c end
        end
    end
    return nil
end

-- профиль открыт и это именно нужный игрок (по нику в окне)
local function profileIs(plr)
    local vp = getVP()
    local nav = path(vp, "Main", "Nav")
    if not (nav and isVisible(nav)) then return false end
    local n1, n2 = plr.Name:lower(), plr.DisplayName:lower()
    for _, d in ipairs(vp:GetDescendants()) do
        if d:IsA("TextLabel") then
            local t = d.Text:lower()
            if (t == n1 or t == n2) and isVisible(d) then return true end
        end
    end
    local ic = path(nav, "Profile", "Icon")
    if ic then
        local id = tostring(ic.Image):match("id=(%d+)")
        if id == tostring(plr.UserId) then return true end
    end
    return false
end

local function validName(t)
    return t ~= "" and t ~= "Loading" and t ~= "..."
end

local function collect(frame)
    local out = {}
    for _, d in ipairs(frame:GetDescendants()) do
        if d.Name == "NewItem" and d:IsA("Frame") then
            local lb = path(d, "ItemName", "Label")
            if lb and lb:IsA("TextLabel") and validName(lb.Text) then
                local n = 1
                local amt = path(d, "Container", "Amount")
                if amt and amt:IsA("TextLabel") and isVisible(amt) then
                    n = tonumber(amt.Text:match("%d+")) or 1
                end
                out[#out + 1] = {name = lb.Text, amount = n}
            end
        end
    end
    return out
end

local function readTab(key)
    local vp = getVP()
    local tb = path(vp, "Main", "Weapons", "TitleBar", "Container", key)
    local function sel()
        local vb = tb and tb:FindFirstChild("ViewBorder")
        return vb and isVisible(vb) or false
    end
    if tb and not sel() then
        clickUntil(tb:FindFirstChild("View"), sel)
    end
    local items, last, stable = {}, -1, 0
    for _ = 1, 40 do
        task.wait(0.07)
        local frame = path(getVP(), "Main", "Weapons", "Items", "Container", key)
        if frame and isVisible(frame) then
            if frame:IsA("ScrollingFrame") then
                pcall(function()
                    frame.CanvasPosition = Vector2.new(0, frame.AbsoluteCanvasSize.Y)
                end)
            end
            items = collect(frame)
            if #items == last then stable += 1 else stable = 0 end
            last = #items
            if stable >= 2 and #items > 0 or stable >= 4 then break end
        end
    end
    local frame = path(getVP(), "Main", "Weapons", "Items", "Container", key)
    if frame and frame:IsA("ScrollingFrame") then
        pcall(function() frame.CanvasPosition = Vector2.new(0, 0) end)
    end
    return items
end

local function openProfile(plr, names)
    local function have() return findLabel(plr, scanLabels(names)) end
    if not have() then
        local plus = path(playerGui, "MainGUI", "Lobby", "LeaderBar", "Main", "Plus", "Button")
        if not clickUntil(plus, have) then return false, "не открылся список игроков" end
    end
    local label = have()
    if not label then return false, "игрока нет в списке" end
    local card = findClickable(label)
    if not card then return false, "нет кнопки у карточки" end
    if not clickUntil(card, function() return findButtonByText("Profile") ~= nil end) then
        return false, "не появилась кнопка Profile"
    end
    local pb = findButtonByText("Profile")
    if not clickUntil(pb, function() return profileIs(plr) end) then
        return false, "профиль не открылся"
    end
    local function wOpen()
        local w = path(getVP(), "Main", "Weapons")
        return w and isVisible(w) or false
    end
    if not wOpen() then
        clickUntil(path(getVP(), "Main", "Nav", "Weapons"), wOpen)
    end
    if not wOpen() then return false, "вкладка Weapons не открылась" end
    task.wait(0.2)
    return true
end

local running, stopFlag = false, false

local function runParser()
    if running then return end
    local q = normalize(input.Text)
    if q == "" then setStatus("Впиши название оружия или число.") return end
    local target = tonumber((q:gsub("[,%s]", "")))
    running, stopFlag = true, false
    clearResults()
    banner.Visible = false

    if target and #texts == 0 then
        setStatus("Жду загрузку цен...")
        for _ = 1, 40 do
            if #texts > 0 then break end
            task.wait(0.5)
        end
        if #texts == 0 then
            setStatus("Цены не загрузились, попробуй позже.")
            running = false
            return
        end
    end

    local list, names = {}, {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then
            list[#list + 1] = p
            names[p.Name:lower()] = true
            names[p.DisplayName:lower()] = true
        end
    end
    if #list == 0 then
        setStatus("На сервере никого нет. Жми «Перезайти».")
        running = false
        return
    end

    cover.Visible = true
    pp.Visible = true

    local hits, skipped = 0, 0
    for i, plr in ipairs(list) do
        if stopFlag then break end
        if not plr.Parent then continue end
        setStatus(string.format("%d/%d: %s | найдено: %d", i, #list, plr.Name, hits))

        local items, err
        for _ = 1, 2 do
            closeProfile()
            local ok, e = pcall(function()
                local o, er = openProfile(plr, names)
                if not o then err = er return end
                local all = {}
                local hit = false
                for _, t in ipairs(TABS) do
                    for _, it in ipairs(readTab(t[1])) do
                        it.tab = t[2]
                        all[#all + 1] = it
                        if not target and normalize(it.name):find(q, 1, true) then hit = true end
                    end
                    if hit then break end
                end
                items = all
            end)
            if not ok then err = tostring(e) end
            if items then break end
        end
        closeProfile()

        if not items then
            skipped += 1
            addResult("⚠ " .. plr.Name .. ": " .. tostring(err), Color3.fromRGB(255, 200, 120))
        elseif target then
            local total = 0
            for _, it in ipairs(items) do
                local v = getValue(it.name)
                if v then total += v * it.amount end
            end
            if total >= target then
                hits += 1
                local msg = "✅ Успешно: " .. plr.Name .. " — " .. fmt(total) .. " валют"
                addResult(msg, COLOR_WIN)
                notify(msg)
                if stopOnHit then stopFlag = true end
            end
        else
            local found = {}
            for _, it in ipairs(items) do
                if normalize(it.name):find(q, 1, true) then
                    found[#found + 1] = it.name .. " [" .. it.tab .. "]" .. (it.amount > 1 and (" x" .. it.amount) or "")
                end
            end
            if #found > 0 then
                hits += 1
                addResult("✅ " .. plr.Name .. ": " .. table.concat(found, ", "), COLOR_WIN)
                notify("✅ НАЙДЕНО у " .. plr.Name .. ": " .. table.concat(found, ", "))
                if stopOnHit then stopFlag = true end
            end
        end
    end

    if hits > 0 then
        setStatus("Успешно! Найдено: " .. hits .. " (пропущено: " .. skipped .. ")")
    elseif stopFlag then
        setStatus("Остановлено. Найдено: 0")
    else
        setStatus("Никого не нашли (пропущено: " .. skipped .. "). Жми «Перезайти».")
        addResult("Никого подходящего на этом сервере.", Color3.fromRGB(255, 160, 160))
    end
    running = false
end
